统一代码模板（栈存索引）

Deque<Integer> st = new ArrayDeque<>();
for (int i = 0; i < n; i++) {
    while (!st.isEmpty() && /* 当前元素 与 栈顶 的大小关系：> 或 < */) {
        int idx = st.pop();   // 弹出并结算 idx 的答案
    }
    st.push(i);               // 自己进栈等答案
}
